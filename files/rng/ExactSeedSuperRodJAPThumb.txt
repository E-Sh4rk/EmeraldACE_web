; Ensure the current map has fishing encounters.

XX ?= (Seed & 0xFF0000)>>16

ZZ?= ((0xB6-YY) & neg)? ( ((0xB9-YY)&neg)?(((0xEE-YY)&neg)?(YY-0xEE):0):(YY-0xB6)):0
VV?= ((0xB6-YY) & neg)? ( ((0xB9-YY)&neg)?(((0xEE-YY)&neg)?(0xEE):YY):(0xB6)):YY

UVdata = Seed ? ((uu<<24)|(UU<<16)|(vv<<8)|VV) : 0x12121212
