@@ title = "Change Roamer PID"
@@ author = "Sleipnir (ported by Adrichu00)"
@@ exit = "GrabACEExit"

; This code can be executed both if you already have an active roamer or not,
; either way the chosen roamer will be active.
; Features such as species and IVs will be the ones from the last active roamer.

PID = 0x1234ABCD

@@

sbc r11,pc,0xBA00
adc r11, r11, 0x28C0
mvn r12,0X3FC
bic r12,r12,0XE2
strb r12, [r11, 0xEE]    %%    Activate roamer
movs r12, {PID} ?
str r12, [r11, 0xDF]!   %%    Store PID in roamer location
