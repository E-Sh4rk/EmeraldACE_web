@@ title = "Respawn Lugia and Ho-Oh"

@@ exit = "GrabACEExit"

; Clears FLAG_FOUGHT_LUGIA and FLAG_LUGIA_FLEW_AWAY or FLAG_FOUGHT_HO_OH and FLAG_HO_OH_FLEW_AWAY, allowing Lugia and Ho-Oh to respawn on Navel Rock

pokemon = 1
; Select 1 for Lugia, 2 for Ho-Oh

flag_mask = pokemon == 1 ? 0x0024 : 0x0048

@@

sub  r11, pc, 0xB1F8 ?
mov  r12, {flag_mask} ?
bic  r0, r12, 0x00
ldrh r12, [r11]
bic  r12, r12, r0
strh r12, [r11]
